   347→async function ensureLocalSeed() {
   348→  const db = await getLocalDB();
   349→  const seeded = await db.meta.get("seeded");
   350→  if (seeded) {
   351→    // Incluso si ya está sembrado, limpiar duplicados
   352→    await cleanDuplicateCategories(db);
   353→    return;
   354→  }
   355→
   356→  // === NUEVA INSTALACIÓN: BASE DE DATOS LIMPIA ===
   357→  // Solo categorías por defecto + cuenta de efectivo
   358→  // NO crear gastos, suscripciones, metas, recordatorios ni presupuestos de ejemplo
   359→
   360→  // Categorías de egreso por defecto
   361→  const cats: LocalCategory[] = [];
   362→  for (const c of DEFAULT_CATEGORIES) {
   363→    cats.push({
   364→      id: localId(),
   365→      name: c.name,
   366→      icon: c.icon,
   367→      color: c.color,
   368→      type: "expense",
   369→      isDefault: true,
   370→      subcategories: [],
   371→      createdAt: new Date().toISOString(),
   372→    });
   373→  }
   374→  // Asegurar "Conveniencia"
   375→  if (!cats.find((c) => c.name === "Conveniencia")) {
   376→    cats.push({
   377→      id: localId(),
   378→      name: "Conveniencia",
   379→      icon: "ShoppingCart",
   380→      color: "amber",
   381→      type: "expense",
   382→      isDefault: true,
   383→      subcategories: [],
   384→      createdAt: new Date().toISOString(),
   385→    });
   386→  }
   387→  // Categorías de ingreso
   388→  for (const ic of DEFAULT_INCOME_CATEGORIES) {
   389→    if (!cats.find((c) => c.name.toLowerCase() === ic.name.toLowerCase())) {
   390→      cats.push({
   391→        id: localId(),
   392→        name: ic.name,
   393→        icon: ic.icon,
   394→        color: ic.color,
   395→        type: "income",
   396→        isDefault: true,
   397→        subcategories: [],
   398→        createdAt: new Date().toISOString(),
   399→      });
   400→    }
   401→  }
   402→  await db.categories.bulkPut(cats);
   403→
   404→  // Única cuenta: Efectivo con saldo $0
   405→  await db.accounts.put({
   406→    id: localId(),
   407→    name: "Efectivo",
   408→    type: "cash",
   409→    balance: 0,
   410→    currency: "MXN",
   411→    color: "emerald",
   412→    isDefault: true,
   413→    createdAt: new Date().toISOString(),
   414→    updatedAt: new Date().toISOString(),
   415→  });
   416→
   417→  await db.meta.put({ key: "seeded", value: true });
   418→}

   420→// Limpiar categorías duplicadas — mantiene solo la primera de cada nombre
   421→async function cleanDuplicateCategories(db: Awaited<ReturnType<typeof getLocalDB>>): Promise<void> {
   422→  const allCats = await db.categories.toArray();
   423→  const seen = new Map<string, LocalCategory>();
   424→  const toDelete: string[] = [];
   425→
   426→  for (const cat of allCats) {
   427→    const key = cat.name.toLowerCase();
   428→    if (seen.has(key)) {
   429→      // Duplicado — eliminar el más reciente
   430→      toDelete.push(cat.id);
   431→    } else {
   432→      seen.set(key, cat);
   433→    }
   434→  }
   435→
   436→  if (toDelete.length > 0) {
   437→    await db.categories.bulkDelete(toDelete);
   438→  }
   439→}

   441→async function resolveExpense(e: LocalExpense): Promise<Expense> {
   442→  const db = await getLocalDB();
   443→  const category = (await db.categories.get(e.categoryId))!;
   444→  const subcategory = e.subcategoryId
   445→    ? category?.subcategories.find((s) => s.id === e.subcategoryId)
   446→    : null;
   447→  const merchant = e.merchantId ? await db.merchants.get(e.merchantId) : undefined;
   448→  const account = e.accountId ? await db.accounts.get(e.accountId) : undefined;
   449→  return {
   450→    ...e,
   451→    category: category || { id: e.categoryId, name: "Otros", icon: "Wallet", color: "slate", type: "expense", isDefault: false, subcategories: [], createdAt: "" },
   452→    subcategory: subcategory ? { id: subcategory.id, name: subcategory.name } : null,
   453→    merchant: merchant ? { id: merchant.id, name: merchant.name } : null,
   454→    account: account || null,
   455→  };
   456→}

   500→  async createSubcategory(categoryId: string, name: string): Promise<{ id: string; name: string; categoryId: string }> {
   501→    const db = await getLocalDB();
   502→    const cat = await db.categories.get(categoryId);
   503→    if (!cat) throw new Error("Categoría no encontrada");
   504→    const trimmed = name.trim();
   505→    if (!trimmed) throw new Error("El nombre es obligatorio");
   506→    if (cat.subcategories.find((s) => s.name.toLowerCase() === trimmed.toLowerCase())) {
   507→      throw new Error("Ya existe una subcategoría con ese nombre");
   508→    }
   509→    const sub = { id: localId(), name: trimmed, categoryId };
   510→    cat.subcategories.push(sub);
   511→    await db.categories.put(cat);
   512→    return sub;
   513→  },
   514→  async updateSubcategory(id: string, name: string): Promise<{ id: string; name: string }> {
   515→    const db = await getLocalDB();
   516→    const trimmed = name.trim();
   517→    if (!trimmed) throw new Error("El nombre no puede estar vacío");
   518→    const cats = await db.categories.toArray();
   519→    for (const cat of cats) {
   520→      const sub = cat.subcategories.find((s) => s.id === id);
   521→      if (sub) {

   894→  async processSubscriptions(): Promise<{ charged: number; reminders: number; advanced: number; details: Array<{ name: string; action: string; amount?: number }> }> {
   895→    const db = await getLocalDB();
   896→    const now = new Date();
   897→    const threeDaysFromNow = new Date(now.getTime() + 3 * 86400000);
   898→    const result = { charged: 0, reminders: 0, advanced: 0, details: [] as Array<{ name: string; action: string; amount?: number }> };
   899→
   900→    const subs = (await db.subscriptions.toArray()).filter((s) => s.active);
   901→    const cats = await db.categories.toArray();
   902→
   903→    for (const sub of subs) {
   904→      let nextDate = new Date(sub.nextDate);
   905→
   906→      // 1. Cobrar suscripciones vencidas
   907→      while (nextDate.getTime() <= now.getTime()) {
   908→        const defaultCat = cats.find((c) => c.name === "Servicios") || cats.find((c) => c.name === "Otros") || cats[0];
   909→        const e: LocalExpense = {
   910→          id: localId(),
   911→          amount: sub.amount,
   912→          type: "expense",
   913→          currency: sub.currency,
   914→          date: new Date(nextDate).toISOString(),
   915→          categoryId: sub.categoryId || defaultCat?.id || "",
   916→          merchantName: sub.merchantName || sub.name,
   917→          paymentMethod: "credit",
   918→          accountId: sub.accountId || null,
   919→          notes: `Suscripción: ${sub.name}`,
   920→          tags: "suscripcion,recurrente",
   921→          isRecurring: true,
   922→          recurringName: sub.name,
   923→          source: "subscription",
   924→          createdAt: now.toISOString(),
   925→          updatedAt: now.toISOString(),
   926→        };
   927→        await db.expenses.put(e);
   928→
   929→        // Actualizar balance
   930→        if (sub.accountId) {
   931→          const acc = await db.accounts.get(sub.accountId);
   932→          if (acc) {
   933→            acc.balance -= sub.amount;
   934→            acc.updatedAt = now.toISOString();
   935→            await db.accounts.put(acc);
   936→          }
   937→        }
   938→
   939→        // Avanzar fecha
   940→        nextDate = advanceDateLocal(nextDate, sub.period);
   941→        result.charged++;
   942→        result.advanced++;
   943→        result.details.push({ name: sub.name, action: "charged", amount: sub.amount });
   944→      }
   945→
   946→      // Actualizar nextDate si avanzó
   947→      if (nextDate.getTime() !== new Date(sub.nextDate).getTime()) {
   948→        sub.nextDate = nextDate.toISOString();
   949→        sub.updatedAt = now.toISOString();
   950→        await db.subscriptions.put(sub);
   951→      }
   952→
   953→      // 2. Crear recordatorio si vence en 3 días
   954→      if (nextDate.getTime() <= threeDaysFromNow.getTime() && nextDate.getTime() > now.getTime()) {
   955→        const existing = await db.reminders.where("dueDate").between(now.toISOString(), nextDate.toISOString(), true, true).toArray();
   956→        const alreadyExists = existing.some((r) => r.title.includes(sub.name));
   957→        if (!alreadyExists) {
   958→          await db.reminders.put({
   959→            id: localId(),
   960→            title: `Suscripción por vencer: ${sub.name}`,
   961→            type: "pay_service",
   962→            dueDate: nextDate.toISOString(),
   963→            done: false,
   964→            notes: `Se cobrarán ${sub.amount} ${sub.currency} de tu suscripción a ${sub.name}.`,
   965→            createdAt: now.toISOString(),
   966→          });
   967→          result.reminders++;
   968→          result.details.push({ name: sub.name, action: "reminder" });
   969→        }
   970→      }
   971→    }
   972→
   973→    return result;
   974→  },

  1046→  async getStats(month: string): Promise<Stats> {
  1047→    await ensureLocalSeed();
  1048→    const db = await getLocalDB();
  1049→    const [y, m] = month.split("-").map(Number);
  1050→    const startISO = new Date(y, m - 1, 1).toISOString();
  1051→    const endISO = new Date(y, m, 0, 23, 59, 59, 999).toISOString();
  1052→    const prevStartISO = new Date(y, m - 2, 1).toISOString();
  1053→    const prevEndISO = new Date(y, m - 1, 0, 23, 59, 59, 999).toISOString();
  1054→
  1055→    const [expenses, prevExpenses, accounts, budgets, subscriptions, goals] = await Promise.all([
  1056→      db.expenses.where("date").between(startISO, endISO, true, true).toArray(),
  1057→      db.expenses.where("date").between(prevStartISO, prevEndISO, true, true).toArray(),
  1058→      db.accounts.toArray(),
  1059→      db.budgets.where("month").equals(month).toArray(),
  1060→      db.subscriptions.toArray().then((subs) => subs.filter((s) => s.active)),
  1061→      db.goals.toArray(),
  1062→    ]);
  1063→
  1064→    const cats = await db.categories.toArray();
  1065→    const catMap = new Map(cats.map((c) => [c.id, c]));
  1066→
  1067→    const expenseList = expenses.filter((e) => e.type !== "income");
  1068→    const incomeList = expenses.filter((e) => e.type === "income");
  1069→
  1070→    const totalSpent = expenseList.reduce((s, e) => s + e.amount, 0);
  1071→    const totalIncome = incomeList.reduce((s, e) => s + e.amount, 0);
  1072→    const prevTotalSpent = prevExpenses.filter((e) => e.type !== "income").reduce((s, e) => s + e.amount, 0);
  1073→    const variation = prevTotalSpent > 0 ? ((totalSpent - prevTotalSpent) / prevTotalSpent) * 100 : 0;
  1074→    const totalBalance = accounts.reduce((s, a) => s + a.balance, 0);
  1075→    const totalBudget = budgets.reduce((s, b) => s + b.amount, 0);
  1076→    const budgetRemaining = totalBudget - totalSpent;
  1077→    const totalSaved = totalIncome - totalSpent;
  1078→
  1079→    // Top categorías (solo egresos)
  1080→    const byCat: Record<string, { name: string; color: string; icon: string; total: number; count: number }> = {};
  1081→    for (const e of expenseList) {
  1082→      const c = catMap.get(e.categoryId);
  1083→      if (!byCat[e.categoryId]) byCat[e.categoryId] = { name: c?.name || "Otros", color: c?.color || "slate", icon: c?.icon || "Wallet", total: 0, count: 0 };
  1084→      byCat[e.categoryId].total += e.amount;
  1085→      byCat[e.categoryId].count += 1;
  1086→    }
  1087→    const topCategories = Object.values(byCat).sort((a, b) => b.total - a.total).slice(0, 5);
  1088→
  1089→    // Por día (solo egresos)
  1090→    const totalDays = daysInMonth(new Date(y, m - 1, 1));
  1091→    const byDay: Array<{ day: number; total: number }> = [];
  1092→    for (let d = 1; d <= totalDays; d++) byDay.push({ day: d, total: 0 });
  1093→    for (const e of expenseList) byDay[new Date(e.date).getDate() - 1].total += e.amount;
  1094→
  1095→    // Por método (solo egresos)
  1096→    const byMethod: Record<string, number> = {};
  1097→    for (const e of expenseList) {
  1098→      const method = e.paymentMethod || "cash";
  1099→      byMethod[method] = (byMethod[method] || 0) + e.amount;
  1100→    }
  1101→
  1102→    // Por comercio (solo egresos)
  1103→    const byMerch: Record<string, number> = {};
  1104→    for (const e of expenseList) {
  1105→      const name = e.merchantName || "Otro";
  1106→      byMerch[name] = (byMerch[name] || 0) + e.amount;
  1107→    }
  1108→    const topMerchants = Object.entries(byMerch).map(([name, total]) => ({ name, total })).sort((a, b) => b.total - a.total).slice(0, 10);
  1109→
  1110→    // Por cuenta (solo egresos)
  1111→    const byAcc: Record<string, { name: string; total: number; color: string }> = {};
  1112→    for (const e of expenseList) {
  1113→      const id = e.accountId || "none";
  1114→      const acc = accounts.find((a) => a.id === e.accountId);
  1115→      const name = acc?.name || "Sin cuenta";
  1116→      const color = acc?.color || "slate";
  1117→      if (!byAcc[id]) byAcc[id] = { name, total: 0, color };
  1118→      byAcc[id].total += e.amount;
  1119→    }
  1120→
  1121→    // Budget usage (solo egresos)
  1122→    const budgetUsage = budgets.map((b) => {
  1123→      const spent = expenseList.filter((e) => e.categoryId === b.categoryId).reduce((s, e) => s + e.amount, 0);
  1124→      const c = catMap.get(b.categoryId);
  1125→      return {
  1126→        id: b.id,
  1127→        categoryId: b.categoryId,
  1128→        categoryName: c?.name || "Otros",
  1129→        categoryColor: c?.color || "slate",
  1130→        categoryIcon: c?.icon || "Wallet",
  1131→        amount: b.amount,
  1132→        spent,
  1133→        remaining: b.amount - spent,
  1134→        percentage: b.amount > 0 ? (spent / b.amount) * 100 : 0,
  1135→      };
  1136→    });
  1137→
  1138→    // Comparación categorías (solo egresos)
  1139→    const prevByCat: Record<string, number> = {};
  1140→    for (const e of prevExpenses.filter((e) => e.type !== "income")) prevByCat[e.categoryId] = (prevByCat[e.categoryId] || 0) + e.amount;
  1141→    const categoryComparison = Object.entries(byCat).map(([catId, c]) => ({
  1142→      ...c,
  1143→      categoryId: catId,
  1144→      prev: prevByCat[catId] || 0,
  1145→      variation: prevByCat[catId] ? ((c.total - prevByCat[catId]) / prevByCat[catId]) * 100 : 0,
  1146→    })).sort((a, b) => b.total - a.total);
  1147→
  1148→    // Promedios
  1149→    const today = new Date();
  1150→    const daysElapsed = today.getMonth() === m - 1 && today.getFullYear() === y ? today.getDate() : totalDays;
  1151→    const avgDaily = daysElapsed > 0 ? totalSpent / daysElapsed : 0;
  1152→    const projectedMonth = avgDaily * totalDays;
  1153→
  1154→    // Recent expenses
  1155→    const sortedExpenses = [...expenses].sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
  1156→    const recentExpenses = await Promise.all(sortedExpenses.slice(0, 6).map(resolveExpense));
  1157→
  1158→    return {
  1159→      month,
  1160→      summary: {
  1161→        totalBalance,
  1162→        totalSpent,
  1163→        totalIncome,
  1164→        prevTotalSpent,
  1165→        variation,
  1166→        totalBudget,
  1167→        budgetRemaining,
  1168→        budgetPercentage: totalBudget > 0 ? (totalSpent / totalBudget) * 100 : 0,
  1169→        totalSaved,
  1170→        monthlyGoal: 0,
  1171→        expenseCount: expenseList.length,
  1172→        incomeCount: incomeList.length,
  1173→        avgDaily,
  1174→        avgWeekly: avgDaily * 7,
  1175→        avgMonthly: totalSpent,
  1176→        projectedMonth,
  1177→        subscriptionsTotal: subscriptions.reduce((s, x) => s + x.amount, 0),
  1178→      },
  1179→      topCategories,
  1180→      byDay,
  1181→      byMethod: Object.entries(byMethod).map(([method, total]) => ({ method, total })),
  1182→      topMerchants,
  1183→      byAccount: Object.values(byAcc),
  1184→      budgetUsage,
  1185→      categoryComparison,
  1186→      goals,
  1187→      recentExpenses,
  1188→    };
  1189→  },

  1191→  async bulkImport(expenses: Array<Record<string, unknown>>): Promise<{ created: number; failed: number; total: number; merchantsCreated: number }> {

  1237→function advanceDateLocal(date: Date, period: string): Date {
  1238→  const d = new Date(date);
  1239→  if (period === "yearly") {
  1240→    d.setFullYear(d.getFullYear() + 1);

  1242→    d.setDate(d.getDate() + 7);
