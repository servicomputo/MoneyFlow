   153→      setHeaders(cols);
   154→      setRows(json);

   156→      // Auto-mapear columnas por nombre
   157→      const autoMapping: Record<string, string> = {};

   159→        const match = cols.find((c) => {
   160→          const cl = c.toLowerCase().trim();
   161→          if (field.key === "amount")
   162→            return ["importe", "amount", "total", "monto", "valor"].includes(cl);
   163→          if (field.key === "date")
   164→            return ["fecha", "date", "día", "dia", "fecha de gasto"].includes(cl);
   165→          if (field.key === "categoryName")
   166→            return ["categoría", "categoria", "category", "categoría de gasto"].includes(cl);
   167→          if (field.key === "subcategoryName")
   168→            return ["subcategoría", "subcategoria", "subcategory", "sub"].includes(cl);
   169→          if (field.key === "merchantName")
   170→            return ["comercio", "merchant", "establecimiento", "tienda", "proveedor"].includes(cl);
   171→          if (field.key === "paymentMethod")
   172→            return ["método de pago", "metodo de pago", "payment method", "pago", "método"].includes(cl);
   173→          if (field.key === "accountName")
   174→            return ["cuenta", "account", "cuenta bancaria", "tarjeta"].includes(cl);
   175→          if (field.key === "notes")
   176→            return ["notas", "notes", "descripción", "descripcion", "detalle", "concepto"].includes(cl);
   177→          if (field.key === "tags")
   178→            return ["etiquetas", "tags", "etiqueta"].includes(cl);
   179→          return false;
   180→        });
   181→        if (match) autoMapping[field.key] = match;
   182→      }
   183→      setMapping(autoMapping);
   184→
   185→      toast.success(`${json.length} filas detectadas`, {
   186→        description: `${cols.length} columnas · ${file.name}`,
   187→      });
   188→    } catch (e) {
   189→      toast.error("No se pudo leer el archivo", {
   190→        description: e instanceof Error ? e.message : "Formato no soportado",
   191→      });
   192→    } finally {
   193→      setParsing(false);
   194→    }

   639→function buildMappedExpenses(
   640→  rows: ParsedRow[],
   641→  mapping: Record<string, string>,
   642→  defaultCategoryId: string
   643→): MappedExpense[] {
   644→  return rows.map((row, idx) => {
   645→    const get = (fieldKey: string) => {
   646→      const col = mapping[fieldKey];
   647→      if (!col) return "";
   648→      const v = row[col];
   649→      return v == null ? "" : String(v).trim();
   650→    };
   651→
   652→    const amountStr = get("amount").replace(/[^0-9.,-]/g, "").replace(/,/g, "");
   653→    const amount = parseFloat(amountStr);
   654→    const dateStr = get("date");
   655→    let date = "";
   656→    if (dateStr) {
   657→      const d = new Date(dateStr);
   658→      if (!isNaN(d.getTime())) date = d.toISOString();
   659→    }
   660→
   661→    let paymentMethod = get("paymentMethod");
   662→    if (paymentMethod) {
   663→      const normalized = paymentMethod.toLowerCase().trim();
   664→      paymentMethod = PAYMENT_METHOD_MAP[normalized] || paymentMethod.toLowerCase();
   665→    }
   666→
   667→    const tagsStr = get("tags");
   668→    const tags = tagsStr
   669→      ? tagsStr.split(/[,;|]/).map((t) => t.trim()).filter(Boolean)
   670→      : [];
   671→
   672→    let _valid = true;
   673→    let _error: string | undefined;
   674→    if (!amount || amount <= 0 || !isFinite(amount)) {
   675→      _valid = false;
   676→      _error = "Importe inválido";
   677→    } else if (!date) {
   678→      _valid = false;
   679→      _error = "Fecha inválida";
   680→    }
   681→
   682→    const categoryId = defaultCategoryId || undefined;
   683→
   684→    return {
   685→      amount,
   686→      date,
   687→      categoryName: get("categoryName") || undefined,
   688→      subcategoryName: get("subcategoryName") || undefined,
   689→      merchantName: get("merchantName") || undefined,
   690→      paymentMethod: paymentMethod || undefined,
   691→      accountName: get("accountName") || undefined,
   692→      notes: get("notes") || undefined,
   693→      tags: tags.length ? tags : undefined,
   694→      categoryId,
   695→      aiCategorized: !!aiCategoryId,
   696→      _valid,
   697→      _error,
   698→    };
   699→  });
   700→}

   702→// Renderiza la celda de categoría en la vista previa:
   703→// - Si viene del archivo, muestra el nombre del archivo
   704→// - Si viene de IA, muestra el nombre resuelto + icono Sparkles
   705→// - Si viene del default, muestra el nombre resuelto
   706→// - Si no hay, muestra "—"
   707→function CategoryCell({
   708→  expense,
   709→  categories,
   710→}: {
   711→  expense: MappedExpense;
   712→  categories?: import("@/lib/data-provider").Category[];
   713→}) {
   714→  if (expense.categoryName) {
   715→    return <span>{expense.categoryName}</span>;
   716→  }
   717→  if (expense.categoryId && categories) {
   718→    const cat = categories.find((c) => c.id === expense.categoryId);
   719→    if (cat) {
   720→      return (
   721→        <span className="inline-flex items-center gap-1">
   722→          {expense.aiCategorized && (
   723→            <Sparkles className="h-3 w-3 text-primary shrink-0" />
   724→          )}
   725→          <span className={expense.aiCategorized ? "text-primary font-medium" : ""}>
   726→            {cat.name}
   727→          </span>
   728→        </span>
   729→      );
   730→    }
   731→  }
   732→  return <span>—</span>;
   733→}
