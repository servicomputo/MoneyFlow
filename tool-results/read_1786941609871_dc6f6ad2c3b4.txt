   637→   637→function buildMappedExpenses(
   638→   638→  rows: ParsedRow[],
   639→   639→  mapping: Record<string, string>,
   640→   640→  defaultCategoryId: string
   641→   641→): MappedExpense[] {
   642→   642→  return rows.map((row) => {
   643→   643→    const get = (fieldKey: string) => {
   644→   644→      const col = mapping[fieldKey];
   645→   645→      if (!col) return "";
   646→   646→      const v = row[col];
   647→   647→      return v == null ? "" : String(v).trim();
   648→   648→    };
   649→   649→
   650→   650→    const amountStr = get("amount").replace(/[^0-9.,-]/g, "").replace(/,/g, "");
   651→   651→    const amount = parseFloat(amountStr);
   652→   652→    const dateStr = get("date");
   653→   653→    let date = "";
   654→   654→    if (dateStr) {
   655→   655→      const d = new Date(dateStr);
   656→   656→      if (!isNaN(d.getTime())) date = d.toISOString();
   657→   657→    }
   658→   658→
   659→   659→    let paymentMethod = get("paymentMethod");
   660→   660→    if (paymentMethod) {
   661→   661→      const normalized = paymentMethod.toLowerCase().trim();
   662→   662→      paymentMethod = PAYMENT_METHOD_MAP[normalized] || paymentMethod.toLowerCase();
   663→   663→    }
   664→   664→
   665→   665→    const tagsStr = get("tags");
   666→   666→    const tags = tagsStr
   667→   667→      ? tagsStr.split(/[,;|]/).map((t) => t.trim()).filter(Boolean)
   668→   668→      : [];
   669→   669→
   670→   670→    let _valid = true;
   671→   671→    let _error: string | undefined;
   672→   672→    if (!amount || amount <= 0 || !isFinite(amount)) {
   673→   673→      _valid = false;
   674→   674→      _error = "Importe inválido";
   675→   675→    } else if (!date) {
   676→   676→      _valid = false;
   677→   677→      _error = "Fecha inválida";
   678→   678→    }
   679→   679→
   680→   680→    return {
   681→   681→      amount,
   682→   682→      date,
   683→   683→      categoryName: get("categoryName") || undefined,
   684→   684→      subcategoryName: get("subcategoryName") || undefined,
   685→   685→      merchantName: get("merchantName") || undefined,
   686→   686→      paymentMethod: paymentMethod || undefined,
   687→   687→      accountName: get("accountName") || undefined,
   688→   688→      notes: get("notes") || undefined,
   689→   689→      tags: tags.length ? tags : undefined,
   690→   690→      categoryId: defaultCategoryId || undefined,
   691→   691→      _valid,
   692→   692→      _error,
   693→   693→    };
   694→   694→  });
   695→   695→}
