    83→    83→function periodBadge(p: string) {
    84→    84→  if (p === "biweekly") return "bg-cyan-500/10 text-cyan-600 dark:text-cyan-400";
    85→    85→  if (p === "yearly") return "bg-purple-500/10 text-purple-600 dark:text-purple-400";

    90→    90→function toMonthly(amount: number, period: string) {
    91→    91→  const factor = PERIODS.find((p) => p.value === period)?.monthsFactor ?? 1;
    92→    92→  return amount * factor;
    93→    93→}

   280→   280→        toast.error("Selecciona las cuentas de origen y destino");

   283→   283→      if (formAccount === formDestAccount) {
   284→   284→        toast.error("Las cuentas de origen y destino deben ser diferentes");
   285→   285→        return;
   286→   286→      }
   287→   287→    }
   288→   288→    setSaving(true);
   289→   289→    try {
   290→   290→      const fromAccount = accounts?.find((a) => a.id === formAccount);
   291→   291→      const toAccount = accounts?.find((a) => a.id === formDestAccount);
   292→   292→      const name =
   293→   293→        txnType === "transfer" && fromAccount && toAccount
   294→   294→          ? formName.trim() || `Transferencia: ${fromAccount.name} → ${toAccount.name}`
   295→   295→          : formName.trim();
   296→   296→
   297→   297→      const payload: Record<string, unknown> = {
   298→   298→        name,
   299→   299→        type: txnType,
   300→   300→        merchantName:
   301→   301→          txnType === "transfer"
   302→   302→            ? toAccount?.name || formMerchant.trim() || undefined
   303→   303→            : formMerchant.trim() || undefined,
   304→   304→        amount,
   305→   305→        currency: "MXN",
   306→   306→        period: formPeriod,
   307→   307→        nextDate: new Date(formNextDate).toISOString(),
   308→   308→        categoryId: txnType === "transfer" ? null : formCategory || undefined,
   309→   309→        accountId: formAccount || undefined,
   310→   310→        active: editing ? editing.active : true,
   311→   311→      };
   312→   312→      if (editing) {
   313→   313→        await mutations.updateSubscription(editing.id, payload);
   314→   314→        toast.success("Transacción recurrente actualizada");
   315→   315→      } else {
   316→   316→        await mutations.createSubscription(payload);
   317→   317→        toast.success(
   318→   318→          txnType === "expense"
   319→   319→            ? "Gasto recurrente creado"
   320→   320→            : txnType === "income"
   321→   321→            ? "Ingreso recurrente creado"
   322→   322→            : "Transferencia recurrente creada"
   323→   323→        );
   324→   324→      }
   325→   325→      qc.invalidateQueries({ queryKey: ["subscriptions"] });
   326→   326→      qc.invalidateQueries({ queryKey: ["stats"] });
   327→   327→      setDialogOpen(false);
   328→   328→    } catch {
   329→   329→      toast.error("No se pudo guardar la transacción recurrente");
   330→   330→    } finally {
   331→   331→      setSaving(false);
   332→   332→    }
   333→   333→  }
