    54→    54→function advanceDate(date: Date, period: string): Date {
    55→    55→  const d = new Date(date);
    56→    56→  if (period === "yearly") {
    57→    57→    d.setFullYear(d.getFullYear() + 1);
    58→    58→  } else if (period === "weekly") {
    59→    59→    d.setDate(d.getDate() + 7);
    60→    60→  } else if (period === "biweekly") {
    61→    61→    d.setDate(d.getDate() + 15);
    62→    62→  } else {
    63→    63→    d.setMonth(d.getMonth() + 1);
    64→    64→  }
    65→    65→  return d;
    66→    66→}
