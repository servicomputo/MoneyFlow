    72→const PERIODS = [
    73→  { value: "weekly", label: "Semanal", monthsFactor: 12 / 52 },
    74→  { value: "biweekly", label: "Quincenal", monthsFactor: 12 / 24 },
    75→  { value: "monthly", label: "Mensual", monthsFactor: 1 },
    76→  { value: "yearly", label: "Anual", monthsFactor: 1 / 12 },
    77→] as const;
    78→
    79→function periodLabel(p: string) {
    80→  return PERIODS.find((x) => x.value === p)?.label || p;
    81→}
    82→
    83→function periodBadge(p: string) {
    84→  if (p === "biweekly") return "bg-cyan-500/10 text-cyan-600 dark:text-cyan-400";
    85→  if (p === "yearly") return "bg-purple-500/10 text-purple-600 dark:text-purple-400";
    86→  if (p === "weekly") return "bg-amber-500/10 text-amber-600 dark:text-amber-400";
    87→  return "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400";
    88→}
    89→
    90→function toMonthly(amount: number, period: string) {
    91→  const factor = PERIODS.find((p) => p.value === period)?.monthsFactor ?? 1;
    92→  return amount * factor;
    93→}
    94→
    95→function toInputDate(d: string) {
    96→  const date = new Date(d);
    97→  if (isNaN(date.getTime())) return "";
    98→  return date.toISOString().slice(0, 10);
    99→}

   102→function TypeIcon({ type, className }: { type: TransactionType; className?: string }) {

   104→  if (type === "transfer") return <ArrowLeftRight className={className} />;
   105→  return <ArrowDownLeft className={className} />;
   106→}

   272→  async function save() {
   273→    const amount = parseFloat(formAmount);
   274→    if (!formName.trim() || isNaN(amount) || amount <= 0 || !formNextDate) {
   275→      toast.error("Completa nombre, monto y fecha del próximo pago");
   276→      return;
   277→    }
   278→    if (txnType === "transfer") {
   279→      if (!formAccount || !formDestAccount) {
   280→        toast.error("Selecciona las cuentas de origen y destino");
   281→        return;
   282→      }
   283→      if (formAccount === formDestAccount) {
   284→        toast.error("Las cuentas de origen y destino deben ser diferentes");
   285→        return;
   286→      }
   287→    }
   288→    setSaving(true);
   289→    try {
   290→      const fromAccount = accounts?.find((a) => a.id === formAccount);
   291→      const toAccount = accounts?.find((a) => a.id === formDestAccount);
   292→      const name =
   293→        txnType === "transfer" && fromAccount && toAccount
   294→          ? formName.trim() || `Transferencia: ${fromAccount.name} → ${toAccount.name}`
   295→          : formName.trim();
   296→
   297→      const payload: Record<string, unknown> = {
   298→        name,
   299→        type: txnType,
   300→        merchantName:
   301→          txnType === "transfer"
   302→            ? toAccount?.name || formMerchant.trim() || undefined
   303→            : formMerchant.trim() || undefined,
   304→        amount,
   305→        currency: "MXN",
   306→        period: formPeriod,
   307→        nextDate: new Date(formNextDate).toISOString(),
   308→        categoryId: txnType === "transfer" ? null : formCategory || undefined,
   309→        accountId: formAccount || undefined,
   310→        active: editing ? editing.active : true,
   311→      };
   312→      if (editing) {
   313→        await mutations.updateSubscription(editing.id, payload);
   314→        toast.success("Transacción recurrente actualizada");
   315→      } else {
   316→        await mutations.createSubscription(payload);
   317→        toast.success(
   318→          txnType === "expense"
   319→            ? "Gasto recurrente creado"
   320→            : txnType === "income"
   321→            ? "Ingreso recurrente creado"
   322→            : "Transferencia recurrente creada"
   323→        );
   324→      }
   325→      qc.invalidateQueries({ queryKey: ["subscriptions"] });
   326→      qc.invalidateQueries({ queryKey: ["stats"] });
   327→      setDialogOpen(false);
   328→    } catch {
   329→      toast.error("No se pudo guardar la transacción recurrente");
   330→    } finally {
   331→      setSaving(false);
   332→    }
   333→  }

   335→  async function toggleActive(s: Subscription, value: boolean) {
   336→    try {
   337→      await mutations.updateSubscription(s.id, { active: value });
   338→      qc.invalidateQueries({ queryKey: ["subscriptions"] });
   339→      qc.invalidateQueries({ queryKey: ["stats"] });
   340→      toast.success(value ? "Transacción activada" : "Transacción pausada");
   341→    } catch {
   342→      toast.error("No se pudo actualizar la transacción");
   343→    }
   344→  }

   346→  async function confirmDelete() {
   347→    if (!toDelete) return;
   348→    setDeleting(true);
   349→    try {
   350→      await mutations.deleteSubscription(toDelete.id);
   351→      toast.success("Transacción eliminada");
   352→      qc.invalidateQueries({ queryKey: ["subscriptions"] });
   353→      qc.invalidateQueries({ queryKey: ["stats"] });
   354→      setToDelete(null);
   355→    } catch {
   356→      // Fallback: marcar inactiva
   357→      try {
   358→        await mutations.updateSubscription(toDelete.id, { active: false });
   359→        qc.invalidateQueries({ queryKey: ["subscriptions"] });
   360→        qc.invalidateQueries({ queryKey: ["stats"] });
   361→        toast.success("Transacción cancelada");
   362→        setToDelete(null);
   363→      } catch {
   364→        toast.error("No se pudo eliminar la transacción");
   365→      }
   366→    } finally {
   367→      setDeleting(false);
   368→    }
   369→  }

   940→
   941→function SubscriptionCard({
   942→  sub,
   943→  onEdit,
   944→  onDelete,
   945→  onToggle,
   946→}: {
   947→  sub: Subscription;
   948→  onEdit: () => void;
   949→  onDelete: () => void;
   950→  onToggle: (v: boolean) => void;
   951→}) {
   952→  const nextDate = new Date(sub.nextDate);
   953→  const daysUntil = Math.ceil(
   954→    (nextDate.getTime() - Date.now()) / (1000 * 60 * 60 * 24)
   955→  );
   956→  const overdue = daysUntil < 0;
   957→  const txnType = normalizeType(sub.type);
   958→  const rt = getRecurringType(txnType);
   959→  const cc = colorClasses(rt.color);
   960→  const LucideIcon = rt.lucideIcon;
   961→  const isTransfer = txnType === "transfer";
   962→  const isIncome = txnType === "income";
   963→
   964→  return (
   965→    <Card
   966→      className={`overflow-hidden hover:shadow-md transition-shadow ${
   967→        !sub.active ? "opacity-60" : ""
   968→      }`}
   969→    >
   970→      <CardContent className="p-5 space-y-4">
   971→        <div className="flex items-start justify-between gap-2">
   972→          <div className="flex items-center gap-3 min-w-0">
   973→            <div className={cn("h-10 w-10 rounded-xl flex items-center justify-center shrink-0", cc.soft, cc.text)}>
   974→              <LucideIcon className="h-5 w-5" />
   975→            </div>
   976→            <div className="min-w-0">
   977→              <p className="font-semibold truncate">{sub.name}</p>
   978→              {sub.merchantName && !isTransfer && (
   979→                <p className="text-xs text-muted-foreground truncate">
   980→                  {sub.merchantName}
   981→                </p>
   982→              )}
   983→              {isTransfer && sub.merchantName && (
   984→                <p className="text-xs text-muted-foreground truncate">
   985→                  → {sub.merchantName}
   986→                </p>
   987→              )}
   988→            </div>
   989→          </div>
   990→          <div className="flex flex-col gap-1 items-end">
   991→            <Badge className={cn("text-[10px]", cc.soft, cc.text)}>
   992→              {rt.label}
   993→            </Badge>
   994→            <Badge className={`text-[10px] ${periodBadge(sub.period)}`}>
   995→              {periodLabel(sub.period)}
   996→            </Badge>
   997→          </div>
   998→        </div>
   999→
  1000→        <div className="flex items-end justify-between">
  1001→          <div>
  1002→            <p
  1003→              className={cn(
  1004→                "text-xl font-bold",
  1005→                isIncome
  1006→                  ? "text-emerald-600 dark:text-emerald-400"
  1007→                  : isTransfer
  1008→                  ? "text-purple-600 dark:text-purple-400"
  1009→                  : "text-red-600 dark:text-red-400"
  1010→              )}
  1011→            >
  1012→              {isIncome ? "+" : isTransfer ? "" : "-"}
  1013→              {formatCurrency(sub.amount, sub.currency)}
  1014→            </p>
  1015→            <p className="text-[11px] text-muted-foreground">
  1016→              {formatCurrency(toMonthly(sub.amount, sub.period))}/mes
  1017→            </p>
  1018→          </div>
  1019→          <div className="text-right">
  1020→            <p className="text-xs text-muted-foreground">Próximo pago</p>
  1021→            <p className="text-xs font-medium">
  1022→              {formatDate(sub.nextDate, "short")}
  1023→            </p>
  1024→            <p
  1025→              className={`text-[10px] mt-0.5 ${
  1026→                overdue
  1027→                  ? "text-red-600 dark:text-red-400"
  1028→                  : daysUntil <= 3
  1029→                  ? "text-amber-600 dark:text-amber-400"
  1030→                  : "text-muted-foreground"
  1031→              }`}
  1032→            >
  1033→              {overdue
  1034→                ? `Hace ${Math.abs(daysUntil)} días`
  1035→                : daysUntil === 0
  1036→                ? "Hoy"
  1037→                : `En ${daysUntil} días`}
  1038→            </p>
  1039→          </div>
  1040→        </div>
  1041→
  1042→        {(sub.account || sub.category) && (
  1043→          <div className="flex flex-wrap gap-2">
  1044→            {sub.account && (
  1045→              <Badge variant="outline" className="text-[10px] gap-1 font-normal">
  1046→                <CreditCard className="h-3 w-3" /> {sub.account.name}
  1047→              </Badge>
  1048→            )}
  1049→            {isTransfer && sub.merchantName && (
  1050→              <Badge variant="outline" className="text-[10px] gap-1 font-normal">
  1051→                <ArrowRight className="h-3 w-3" /> {sub.merchantName}
  1052→              </Badge>
  1053→            )}
