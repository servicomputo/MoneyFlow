   312→   312→function CategoryCard({
   313→   313→  category,
   314→   314→  onUpdate,
   315→   315→  onDelete,
   316→   316→  onAddSubcategory,
   317→   317→  onDeleteSubcategory,
   318→   318→  onRenameSubcategory,
   319→   319→}: {
   320→   320→  category: Category;
   321→   321→  onUpdate: (id: string, payload: { name: string; icon: string; color: string; type: string }) => void;
   322→   322→  onDelete: (id: string, name: string) => void;
   323→   323→  onAddSubcategory: (categoryId: string, name: string) => void;
   324→   324→  onDeleteSubcategory: (id: string, name: string) => void;
   325→   325→  onRenameSubcategory: (id: string, name: string) => void;
   326→   326→}) {
   327→   327→  const isIncome = category.type === "income";
   328→   328→  const [editOpen, setEditOpen] = React.useState(false);
   329→   329→  const [deleteOpen, setDeleteOpen] = React.useState(false);
   330→   330→  const [newSub, setNewSub] = React.useState("");
   331→   331→  const [addingSub, setAddingSub] = React.useState(false);
   332→   332→  const [editingSubId, setEditingSubId] = React.useState<string | null>(null);
   333→   333→  const [editingSubName, setEditingSubName] = React.useState("");
   334→   334→
   335→   335→  async function submitNewSub() {
   336→   336→    const name = newSub.trim();
   337→   337→    if (!name) return;
   338→   338→    setAddingSub(true);
   339→   339→    await onAddSubcategory(category.id, name);
   340→   340→    setAddingSub(false);
   341→   341→    setNewSub("");
   342→   342→  }
   343→   343→
   344→   344→  async function submitRenameSub() {
   345→   345→    const name = editingSubName.trim();
   346→   346→    if (!name || !editingSubId) {
   347→   347→      setEditingSubId(null);
   348→   348→      return;
   349→   349→    }
   350→   350→    await onRenameSubcategory(editingSubId, name);
   351→   351→    setEditingSubId(null);
   352→   352→    setEditingSubName("");
   353→   353→  }
   354→   354→
   355→   355→  return (
   356→   356→    <AccordionItem
   357→   357→      value={category.id}
   358→   358→      className="rounded-xl border bg-card shadow-sm overflow-hidden border-b"
   359→   359→    >
   360→   360→      <div className="flex items-start gap-3 p-4">
   361→   361→        <CategoryIcon
   362→   362→          icon={category.icon}
   363→   363→          color={category.color}
   364→   364→          size="lg"
   365→   365→          className="shrink-0"
   366→   366→        />
   367→   367→        <div className="flex-1 min-w-0">
   368→   368→          <div className="flex items-center justify-between gap-2">
   369→   369→            <p className="font-semibold truncate">{category.name}</p>
   370→   370→            <div className="flex items-center gap-0.5">
   371→   371→              <button
   372→   372→                type="button"
   373→   373→                onClick={() => setEditOpen(true)}
   374→   374→                className="text-muted-foreground hover:text-foreground transition-colors p-1 rounded-md hover:bg-accent"
   375→   375→                aria-label="Editar categoría"
   376→   376→                title="Editar"
   377→   377→              >
   378→   378→                <Pencil className="h-3.5 w-3.5" />
   379→   379→              </button>
   380→   380→              <button
   381→   381→                type="button"
   382→   382→                onClick={() => setDeleteOpen(true)}
   383→   383→                className="text-muted-foreground hover:text-red-500 transition-colors p-1 rounded-md hover:bg-accent"
   384→   384→                aria-label="Eliminar categoría"
   385→   385→                title="Eliminar"
   386→   386→              >
   387→   387→                <Trash2 className="h-3.5 w-3.5" />
   388→   388→              </button>
   389→   389→            </div>
   390→   390→          </div>
   391→   391→          <div className="flex items-center gap-2 mt-1.5 flex-wrap">
   392→   392→            <Badge
   393→   393→              variant="secondary"
   394→   394→              className={cn(
   395→   395→                "gap-1",
   396→   396→                isIncome
   397→   397→                  ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 hover:bg-emerald-500/10"
   398→   398→                  : "bg-rose-500/10 text-rose-600 dark:text-rose-400 hover:bg-rose-500/10"
   399→   399→              )}
   400→   400→            >
   401→   401→              {isIncome ? (
   402→   402→                <ArrowDownRight className="h-3 w-3" />
   403→   403→              ) : (
   404→   404→                <ArrowUpRight className="h-3 w-3" />
   405→   405→              )}
   406→   406→              {isIncome ? "Ingreso" : "Gasto"}
   407→   407→            </Badge>
   408→   408→            <Badge variant="outline" className="gap-1">
   409→   409→              <Hash className="h-3 w-3" />
   410→   410→              {category.subcategories.length} sub
   411→   411→            </Badge>
   412→   412→          </div>
   413→   413→        </div>
   414→   414→      </div>
   415→   415→      <AccordionTrigger className="px-4 pb-3 pt-0 hover:no-underline text-xs text-muted-foreground">
   416→   416→        <span>
   417→   417→          {category.subcategories.length > 0
   418→   418→            ? "Ver subcategorías"
   419→   419→            : "Sin subcategorías"}
   420→   420→        </span>
   421→   421→      </AccordionTrigger>
   422→   422→      <AccordionContent className="px-4 pb-4">
   423→   423→        <div className="space-y-2">
   424→   424→          {category.subcategories.length === 0 && (
   425→   425→            <p className="text-xs text-muted-foreground py-2">
   426→   426→              Aún no hay subcategorías.
   427→   427→            </p>
   428→   428→          )}
   429→   429→          {category.subcategories.map((s) => (
   430→   430→            <div
   431→   431→              key={s.id}
   432→   432→              className="flex items-center gap-2 rounded-md border bg-muted/30 px-2.5 py-1.5 text-sm"
   433→   433→            >
   434→   434→              <ChevronRight className="h-3 w-3 text-muted-foreground" />
   435→   435→              {editingSubId === s.id ? (
   436→   436→                <Input
   437→   437→                  value={editingSubName}
   438→   438→                  onChange={(e) => setEditingSubName(e.target.value)}
   439→   439→                  onKeyDown={(e) => {
   440→   440→                    if (e.key === "Enter") submitRenameSub();
   441→   441→                    if (e.key === "Escape") setEditingSubId(null);
   442→   442→                  }}
   443→   443→                  className="h-6 text-sm px-1.5 py-0"
   444→   444→                  autoFocus
   445→   445→                />
   446→   446→              ) : (
   447→   447→                <span className="flex-1 truncate">{s.name}</span>
   448→   448→              )}
   449→   449→              {editingSubId === s.id ? (
   450→   450→                <>
   451→   451→                  <button
   452→   452→                    type="button"
   453→   453→                    onClick={submitRenameSub}
   454→   454→                    className="text-emerald-600 hover:text-emerald-700 p-1"
   455→   455→                    aria-label="Guardar"
   456→   456→                  >
   457→   457→                    <Check className="h-3.5 w-3.5" />
   458→   458→                  </button>
   459→   459→                  <button
   460→   460→                    type="button"
   461→   461→                    onClick={() => setEditingSubId(null)}
   462→   462→                    className="text-muted-foreground hover:text-foreground p-1"
   463→   463→                    aria-label="Cancelar"
   464→   464→                  >
   465→   465→                    <X className="h-3.5 w-3.5" />
   466→   466→                  </button>
   467→   467→                </>
   468→   468→              ) : (
   469→   469→                <>
   470→   470→                  <button
   471→   471→                    type="button"
   472→   472→                    onClick={() => {
   473→   473→                      setEditingSubId(s.id);
   474→   474→                      setEditingSubName(s.name);
   475→   475→                    }}
   476→   476→                    className="text-muted-foreground hover:text-foreground p-1"
   477→   477→                    aria-label="Renombrar"
   478→   478→                    title="Renombrar"
   479→   479→                  >
   480→   480→                    <Pencil className="h-3 w-3" />
   481→   481→                  </button>
   482→   482→                  <button
   483→   483→                    type="button"
   484→   484→                    onClick={() => onDeleteSubcategory(s.id, s.name)}
   485→   485→                    className="text-muted-foreground hover:text-red-500 p-1"
   486→   486→                    aria-label="Eliminar subcategoría"
   487→   487→                    title="Eliminar"
