    54→function advanceDate(date: Date, period: string): Date {
    55→  const d = new Date(date);
    56→  if (period === "yearly") {
    57→    d.setFullYear(d.getFullYear() + 1);
    58→  } else if (period === "weekly") {
    59→    d.setDate(d.getDate() + 7);
    60→  } else if (period === "biweekly") {
    61→    d.setDate(d.getDate() + 15);
    62→  } else {
    63→    d.setMonth(d.getMonth() + 1);
    64→  }
    65→  return d;
    66→}

    72→function FieldRow({
    73→  icon,
    74→  label,
    75→  children,
    76→  divider = true,
    77→  selectedValue,
    78→  placeholder,
    79→  onClick,
    80→  rightIcon,
    81→}: {
    82→  icon: React.ReactNode;
    83→  label: string;
    84→  children?: React.ReactNode;
    85→  divider?: boolean;
    86→  selectedValue?: string;
    87→  placeholder?: string;
    88→  onClick?: () => void;
    89→  rightIcon?: React.ReactNode;
    90→}) {
    91→  const content = (
    92→    <>
    93→      <div className="h-9 w-9 rounded-lg bg-muted/60 flex items-center justify-center shrink-0 text-muted-foreground">
    94→        {icon}
    95→      </div>
    96→      <div className="flex-1 min-w-0">
    97→        <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
    98→          {label}
    99→        </p>
   100→        {children ? (
   101→          <div className="mt-0.5">{children}</div>
   102→        ) : (
   103→          <p className={cn(
   104→            "text-base truncate mt-0.5",
   105→            !selectedValue && "text-muted-foreground/60"
   106→          )}>
   107→            {selectedValue || placeholder || "Selecciona"}
   108→          </p>
   109→        )}
   110→      </div>
   111→    </>
   112→  );
   113→
   114→  if (onClick) {
   115→    return (
   116→      <button
   117→        type="button"
   118→        onClick={onClick}
   119→        className={cn(
   120→          "w-full flex items-center gap-3 py-3.5 text-left transition-colors hover:bg-accent/40 px-2 -mx-2 rounded-lg",
   121→          divider && "border-b border-border/60"
   122→        )}
   123→      >
   124→        {content}
   125→        {rightIcon !== null && (rightIcon || <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />)}
   126→      </button>
   127→    );
   128→  }
   129→
   130→  return (
   131→    <div className={cn(
   132→      "flex items-start gap-3 py-3 px-2 -mx-2 rounded-lg",
   133→      divider && "border-b border-border/60"
   134→    )}>
   135→      {content}
   136→    </div>
   137→  );
   138→}

   554→                    }}
   555→                    className="border-0 px-0 h-auto py-0 text-base focus-visible:ring-0 bg-transparent"
   556→                  />
   557→                  <Button type="button" variant="ghost" size="icon" className="h-7 w-7 shrink-0" onClick={addTag}>
   558→                    <Plus className="h-4 w-4" />
   559→                  </Button>
   560→                </div>
   561→                {tags.length > 0 && (
   562→                  <div className="flex flex-wrap gap-1.5 mt-2">
   563→                    {tags.map((t) => (
   564→                      <Badge key={t} variant="secondary" className="gap-1">
   565→                        {t}
   566→                        <button onClick={() => setTags(tags.filter((x) => x !== t))}>
   567→                          <X className="h-3 w-3" />
   568→                        </button>
   569→                      </Badge>
   570→                    ))}
   571→                  </div>
   572→                )}
   573→              </FieldRow>
   574→
   575→              {/* Notas */}

   577→                <Textarea
   578→                  placeholder="Detalles adicionales..."
   579→                  value={notes}
   580→                  onChange={(e) => setNotes(e.target.value)}

   582→                  className="border-0 px-0 text-base focus-visible:ring-0 bg-transparent resize-none"
   583→                />
   584→              </FieldRow>
   585→
   586→              {/* ¿Es recurrente? */}
   587→              <div className={cn(
   588→                "mt-2 rounded-xl border p-3 flex items-center justify-between gap-3",
   589→                recurrente && (type === "income" ? "border-emerald-500/30 bg-emerald-500/5" : "border-red-500/30 bg-red-500/5")
   590→              )}>
   591→                <div className="flex items-center gap-2.5 min-w-0">
   592→                  <div className={cn(
   593→                    "h-8 w-8 rounded-lg flex items-center justify-center shrink-0",
   594→                    type === "income" ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400" : "bg-red-500/10 text-red-600 dark:text-red-400"
   595→                  )}>
   596→                    <Repeat className="h-4 w-4" />
   597→                  </div>
   598→                  <div className="min-w-0">
   599→                    <p className="text-sm font-medium leading-tight">¿Es recurrente?</p>
   600→                    <p className="text-xs text-muted-foreground leading-tight mt-0.5">
   601→                      Crea un cargo recurrente
   602→                    </p>
   603→                  </div>
   604→                </div>
   605→                <Switch checked={recurrente} onCheckedChange={setRecurrente} aria-label="¿Es recurrente?" />
   606→              </div>
   607→
   608→              {recurrente && (
   609→                <div className="rounded-xl border bg-muted/30 p-3 space-y-2">
   610→                  <div className="flex items-center justify-between">
   611→                    <span className="text-xs text-muted-foreground">Periodicidad</span>
   612→                    <Select value={periodicidad} onValueChange={setPeriodicidad}>
   613→                      <SelectTrigger className="h-8 w-32 text-xs">
   614→                        <SelectValue />
   615→                      </SelectTrigger>
   616→                      <SelectContent>
   617→                        {PERIODS.map((p) => (
   618→                          <SelectItem key={p.value} value={p.value}>{p.label}</SelectItem>
   619→                        ))}
   620→                      </SelectContent>
   621→                    </Select>
   622→                  </div>
   623→                  <p className="text-[11px] text-muted-foreground">
   624→                    Próximo cobro: {advanceDate(date, periodicidad).toLocaleDateString("es-MX", { day: "2-digit", month: "short", year: "numeric" })}
   625→                  </p>
   626→                </div>
   627→              )}
   628→            </div>
   629→          </div>
   630→
   631→          {/* Footer sticky con botón Guardar */}
   632→          <div className="sticky bottom-0 bg-background/95 backdrop-blur border-t p-4">
   633→            <Button
   634→              onClick={handleSave}
   635→              disabled={saveDisabled}
   636→              className={cn(
   637→                "w-full h-12 gap-2 text-base font-semibold",
   638→                type === "income"
   639→                  ? "bg-emerald-600 hover:bg-emerald-700 text-white"
   640→                  : "bg-red-600 hover:bg-red-700 text-white"
   641→              )}
   642→            >
   643→              {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Check className="h-5 w-5" />}
   644→              {saveText}
   645→            </Button>
   646→          </div>
   647→      </ModalContainer>
   648→
   649→      {/* ====== BottomSheets para selectores (aislados, sin empalme de scroll) ====== */}
   650→
   651→      {/* Categoría */}
   652→      <BottomSheet
   653→        open={categorySheetOpen}
   654→        onOpenChange={setCategorySheetOpen}
   655→        title="Selecciona categoría"
   656→      >
   657→        <div className="grid grid-cols-2 gap-2">
   658→          {filteredCategories.map((c) => (
   659→            <button
   660→              key={c.id}
   661→              type="button"
   662→              onClick={() => {

   664→                setSubcategoryId("");
   665→                setCategorySheetOpen(false);
   666→              }}
   667→              className={cn(
   668→                "flex items-center gap-2.5 p-3 rounded-xl border text-left transition-all",
   669→                categoryId === c.id
   670→                  ? "border-primary bg-primary/5 ring-1 ring-primary/20"
   671→                  : "border-border hover:bg-accent/40"
   672→              )}
   673→            >
   674→              <CategoryIcon icon={c.icon} color={c.color} size="sm" className="h-8 w-8" />
   675→              <span className="text-sm font-medium truncate">{c.name}</span>
   676→              {categoryId === c.id && <Check className="h-4 w-4 text-primary ml-auto shrink-0" />}
   677→            </button>
   678→          ))}
   679→        </div>
   680→      </BottomSheet>
   681→
   682→      {/* Subcategoría */}
   683→      <BottomSheet
   684→        open={subCategorySheetOpen}
   685→        onOpenChange={setSubCategorySheetOpen}
   686→        title="Selecciona subcategoría"
   687→      >
   688→        <div className="space-y-1">
   689→          {subcategories.map((s) => (
   690→            <SheetOption
   691→              key={s.id}
   692→              label={s.name}
   693→              selected={subcategoryId === s.id}
   694→              onClick={() => {
   695→                setSubcategoryId(s.id);
   696→                setSubCategorySheetOpen(false);
   697→              }}
   698→            />
   699→          ))}
   700→        </div>
   701→      </BottomSheet>
   702→
   703→      {/* Fecha */}
   704→      <BottomSheet
   705→        open={dateSheetOpen}
   706→        onOpenChange={setDateSheetOpen}
   707→        title="Selecciona fecha"
   708→        maxWidth="sm:max-w-sm"
   709→      >
   710→        <div className="flex justify-center">
   711→          <Calendar
   712→            mode="single"
   713→            selected={date}
   714→            onSelect={(d) => {
   715→              if (d) {
   716→                setDate(d);
   717→                setDateSheetOpen(false);
   718→              }
   719→            }}
   720→            initialFocus
   721→          />
   722→        </div>
   723→      </BottomSheet>
   724→
   725→      {/* Método de pago */}
   726→      <BottomSheet
   727→        open={methodSheetOpen}
   728→        onOpenChange={setMethodSheetOpen}
   729→        title="Selecciona método de pago"
   730→      >
   731→        <div className="space-y-1">
   732→          {PAYMENT_METHODS.map((m) => (
   733→            <SheetOption
   734→              key={m.value}
   735→              icon={<CreditCard className="h-4 w-4" />}
   736→              label={m.label}
   737→              selected={paymentMethod === m.value}
   738→              onClick={() => {
   739→                setPaymentMethod(m.value);
   740→                setMethodSheetOpen(false);
   741→              }}
   742→            />
   743→          ))}
   744→        </div>
   745→      </BottomSheet>
   746→
   747→      {/* Cuenta */}
   748→      <BottomSheet
   749→        open={accountSheetOpen}
   750→        onOpenChange={setAccountSheetOpen}
   751→        title="Selecciona cuenta"
   752→      >
   753→        <div className="space-y-1">
   754→          {accounts?.map((a) => (
   755→            <SheetOption
   756→              key={a.id}
   757→              icon={<Wallet className="h-4 w-4" />}
   758→              label={a.name}
   759→              sublabel={formatCurrency(a.balance)}
   760→              selected={accountId === a.id}
   761→              onClick={() => {
   762→                setAccountId(a.id);
   763→                setAccountSheetOpen(false);
   764→              }}
   765→            />
   766→          ))}
   767→        </div>
   768→      </BottomSheet>
   769→    </>
