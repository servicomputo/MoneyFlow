   152→      setRows(json);

   154→      // Auto-mapear columnas por nombre
   155→      const autoMapping: Record<string, string> = {};

   157→        const match = cols.find((c) => {
   158→          const cl = c.toLowerCase().trim();
   159→          if (field.key === "amount")
   160→            return ["importe", "amount", "total", "monto", "valor"].includes(cl);
   161→          if (field.key === "date")
   162→            return ["fecha", "date", "día", "dia", "fecha de gasto"].includes(cl);
   163→          if (field.key === "categoryName")
   164→            return ["categoría", "categoria", "category", "categoría de gasto"].includes(cl);
   165→          if (field.key === "subcategoryName")
   166→            return ["subcategoría", "subcategoria", "subcategory", "sub"].includes(cl);
   167→          if (field.key === "merchantName")
   168→            return ["comercio", "merchant", "establecimiento", "tienda", "proveedor"].includes(cl);
   169→          if (field.key === "paymentMethod")
   170→            return ["método de pago", "metodo de pago", "payment method", "pago", "método"].includes(cl);
   171→          if (field.key === "accountName")
   172→            return ["cuenta", "account", "cuenta bancaria", "tarjeta"].includes(cl);
   173→          if (field.key === "notes")
   174→            return ["notas", "notes", "descripción", "descripcion", "detalle", "concepto"].includes(cl);
   175→          if (field.key === "tags")
   176→            return ["etiquetas", "tags", "etiqueta"].includes(cl);
   177→          return false;
   178→        });
   179→        if (match) autoMapping[field.key] = match;
   180→      }
   181→      setMapping(autoMapping);
   182→
   183→      toast.success(`${json.length} filas detectadas`, {
   184→        description: `${cols.length} columnas · ${file.name}`,
   185→      });
   186→    } catch (e) {
   187→      toast.error("No se pudo leer el archivo", {
   188→        description: e instanceof Error ? e.message : "Formato no soportado",
   189→      });
   190→    } finally {
   191→      setParsing(false);
   192→    }

   637→function buildMappedExpenses(
   638→  rows: ParsedRow[],
   639→  mapping: Record<string, string>,
   640→  defaultCategoryId: string
   641→): MappedExpense[] {
   642→  return rows.map((row) => {
   643→    const get = (fieldKey: string) => {
   644→      const col = mapping[fieldKey];
   645→      if (!col) return "";
   646→      const v = row[col];
   647→      return v == null ? "" : String(v).trim();
   648→    };
   649→
   650→    const amountStr = get("amount").replace(/[^0-9.,-]/g, "").replace(/,/g, "");
   651→    const amount = parseFloat(amountStr);
   652→    const dateStr = get("date");
   653→    let date = "";
   654→    if (dateStr) {
   655→      const d = new Date(dateStr);
   656→      if (!isNaN(d.getTime())) date = d.toISOString();
   657→    }
   658→
   659→    let paymentMethod = get("paymentMethod");
   660→    if (paymentMethod) {
   661→      const normalized = paymentMethod.toLowerCase().trim();
   662→      paymentMethod = PAYMENT_METHOD_MAP[normalized] || paymentMethod.toLowerCase();
   663→    }
   664→
   665→    const tagsStr = get("tags");
   666→    const tags = tagsStr
   667→      ? tagsStr.split(/[,;|]/).map((t) => t.trim()).filter(Boolean)
   668→      : [];
   669→
   670→    let _valid = true;
   671→    let _error: string | undefined;
   672→    if (!amount || amount <= 0 || !isFinite(amount)) {
   673→      _valid = false;
   674→      _error = "Importe inválido";
   675→    } else if (!date) {
   676→      _valid = false;
   677→      _error = "Fecha inválida";
   678→    }
   679→
   680→    return {
   681→      amount,
   682→      date,
   683→      categoryName: get("categoryName") || undefined,
   684→      subcategoryName: get("subcategoryName") || undefined,
   685→      merchantName: get("merchantName") || undefined,
   686→      paymentMethod: paymentMethod || undefined,
   687→      accountName: get("accountName") || undefined,
   688→      notes: get("notes") || undefined,
   689→      tags: tags.length ? tags : undefined,
   690→      categoryId: defaultCategoryId || undefined,
   691→      _valid,
   692→      _error,
   693→    };
   694→  });
   695→}
