    60→    60→    60→  { value: "weekly", label: "Semanal", monthsFactor: 12 / 52 },

    65→    65→    65→function periodLabel(p: string) {

    69→    69→    69→function periodBadge(p: string) {
    70→    70→    70→  if (p === "yearly") return "bg-purple-500/10 text-purple-600 dark:text-purple-400";
    71→    71→    71→  if (p === "weekly") return "bg-amber-500/10 text-amber-600 dark:text-amber-400";

    73→    73→    73→}
    74→    74→    74→
    75→    75→    75→function toMonthly(amount: number, period: string) {
    76→    76→    76→  const factor = PERIODS.find((p) => p.value === period)?.monthsFactor ?? 1;
    77→    77→    77→  return amount * factor;
    78→    78→    78→}

    80→    80→    80→function toInputDate(d: string) {
    81→    81→    81→  const date = new Date(d);
    82→    82→    82→  if (isNaN(date.getTime())) return "";
    83→    83→    83→  return date.toISOString().slice(0, 10);
    84→    84→    84→}
