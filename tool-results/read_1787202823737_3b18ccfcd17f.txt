    73→export function CategoriesView() {
    74→  const { data: categories, isLoading } = useCategories();
    75→  const queryClient = useQueryClient();
    76→  const [open, setOpen] = React.useState(false);
    77→  const [typeFilter, setTypeFilter] = React.useState<"all" | "expense" | "income">(
    78→    "all"
    79→  );
    80→
    81→  // Botón "+" contextual: abre el diálogo de crear categoría
    82→  useViewAddHandler(() => setOpen(true));
    83→
    84→  async function handleCreate(payload: {
    85→    name: string;
    86→    icon: string;
    87→    color: string;
    88→    type: string;
    89→  }) {
    90→    try {
    91→      await mutations.createCategory(payload);
    92→      await queryClient.invalidateQueries({ queryKey: ["categories"] });
    93→      toast.success("Categoría creada con éxito");
    94→      setOpen(false);
    95→    } catch {
    96→      toast.error("No se pudo crear la categoría");
    97→    }
    98→  }
    99→
   100→  async function handleUpdate(id: string, payload: { name: string; icon: string; color: string; type: string }) {
   101→    try {
   102→      await mutations.updateCategory(id, payload);
   103→      await queryClient.invalidateQueries({ queryKey: ["categories"] });
   104→      toast.success("Categoría actualizada");
   105→    } catch (e) {
   106→      toast.error("No se pudo actualizar", {
   107→        description: e instanceof Error ? e.message : undefined,
   108→      });
   109→    }
   110→  }
   111→
   112→  async function handleDelete(id: string, name: string) {
   113→    try {
   114→      await mutations.deleteCategory(id);
   115→      await queryClient.invalidateQueries({ queryKey: ["categories"] });
   116→      toast.success(`Categoría "${name}" eliminada`);
   117→    } catch (e) {
   118→      toast.error("No se pudo eliminar", {
   119→        description: e instanceof Error ? e.message : undefined,
   120→      });
   121→    }
   122→  }
   123→
   124→  async function handleAddSubcategory(categoryId: string, name: string) {
   125→    try {
   126→      await mutations.createSubcategory(categoryId, name);
   127→      await queryClient.invalidateQueries({ queryKey: ["categories"] });
   128→      toast.success(`Subcategoría "${name}" agregada`);
   129→    } catch (e) {
   130→      toast.error("No se pudo agregar la subcategoría", {
   131→        description: e instanceof Error ? e.message : undefined,
   132→      });
   133→    }
   134→  }
   135→
   136→  async function handleDeleteSubcategory(id: string, name: string) {
   137→    try {
   138→      await mutations.deleteSubcategory(id);
   139→      await queryClient.invalidateQueries({ queryKey: ["categories"] });
   140→      toast.success(`Subcategoría "${name}" eliminada`);
   141→    } catch (e) {
   142→      toast.error("No se pudo eliminar", {
   143→        description: e instanceof Error ? e.message : undefined,
   144→      });
   145→    }
   146→  }
   147→
   148→  async function handleRenameSubcategory(id: string, name: string) {
   149→    try {
   150→      await mutations.updateSubcategory(id, name);
   151→      await queryClient.invalidateQueries({ queryKey: ["categories"] });
   152→      toast.success("Subcategoría renombrada");
   153→    } catch (e) {
   154→      toast.error("No se pudo renombrar", {
   155→        description: e instanceof Error ? e.message : undefined,
   156→      });
   157→    }
   158→  }
   159→
   160→  if (isLoading) return <CategoriesSkeleton />;
   161→
   162→  const list = categories ?? [];
   163→  const filtered = list.filter((c) => typeFilter === "all" || c.type === typeFilter);
   164→  const expenseCount = list.filter((c) => c.type === "expense").length;
   165→  const incomeCount = list.filter((c) => c.type === "income").length;
   166→  const totalSubs = list.reduce((s, c) => s + c.subcategories.length, 0);
   167→
   168→  return (
   169→    <div className="space-y-6">
   170→      {/* Toolbar */}
   171→      <div className="flex flex-wrap items-center justify-between gap-3">
   172→        <div>
   173→          <h2 className="text-lg font-semibold">Tus categorías</h2>
   174→          <p className="text-sm text-muted-foreground">
   175→            Organiza tus gastos e ingresos con categorías personalizadas.
   176→          </p>
   177→        </div>
   178→        <div className="flex items-center gap-2">
   179→          <Select
   180→            value={typeFilter}
   181→            onValueChange={(v) => setTypeFilter(v as typeof typeFilter)}
   182→          >
   183→            <SelectTrigger className="w-[150px]">
   184→              <SelectValue placeholder="Filtrar" />
   185→            </SelectTrigger>
   186→            <SelectContent>
   187→              <SelectItem value="all">Todas</SelectItem>
   188→              <SelectItem value="expense">Gastos</SelectItem>
   189→              <SelectItem value="income">Ingresos</SelectItem>
   190→            </SelectContent>
   191→          </Select>
   192→          <AddCategoryDialog
   193→            open={open}
   194→            onOpenChange={setOpen}
   195→            onSubmit={handleCreate}
   196→          />
   197→        </div>
   198→      </div>
   199→
   200→      {/* Grid */}
   201→      {filtered.length === 0 ? (
   202→        <EmptyState onAdd={() => setOpen(true)} />
   203→      ) : (
   204→        <Accordion
   205→          type="multiple"
   206→          className="grid gap-3 sm:grid-cols-2 xl:grid-cols-3 items-start"
   207→        >
   208→          {filtered.map((cat) => (
   209→            <CategoryCard
   210→              key={cat.id}
   211→              category={cat}
   212→              onUpdate={handleUpdate}
   213→              onDelete={handleDelete}
   214→              onAddSubcategory={handleAddSubcategory}
   215→              onDeleteSubcategory={handleDeleteSubcategory}
   216→              onRenameSubcategory={handleRenameSubcategory}
   217→            />
   218→          ))}
   219→        </Accordion>
   220→      )}
   221→
   222→      {/* Defaults reference */}
   223→      <div className="rounded-xl border bg-muted/30 p-4">
   224→        <div className="flex items-center gap-2 mb-3">
   225→          <Tags className="h-4 w-4 text-primary" />
   226→          <h3 className="text-sm font-semibold">Categorías sugeridas</h3>
   227→          <Badge variant="secondary" className="text-[10px]">
   228→            {DEFAULT_CATEGORIES.length}
   229→          </Badge>
   230→        </div>
   231→        <div className="flex flex-wrap gap-2">
   232→          {DEFAULT_CATEGORIES.map((c) => {
   233→            const cc = colorClasses(c.color);
   234→            return (
   235→              <span
   236→                key={c.name}
   237→                className={cn(
   238→                  "inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-medium",
   239→                  cc.soft,
   240→                  cc.text
   241→                )}
   242→              >
   243→                {renderIcon(CATEGORY_ICONS[c.icon] ?? CATEGORY_ICONS.Wallet, "h-3.5 w-3.5")}
   244→                {c.name}
   245→              </span>
   246→            );
   247→          })}
   248→        </div>
   249→      </div>
   250→
   251→      {/* Summary (totales al final) */}
   252→      <div className="grid gap-4 md:grid-cols-4">
   253→        <SummaryCard
   254→          label="Categorías"
   255→          value={list.length}
   256→          hint="Totales"
   257→          accent="from-emerald-500 to-teal-600"
   258→        />
   259→        <SummaryCard
   260→          label="Gastos"
   261→          value={expenseCount}
   262→          hint="Tipos de gasto"
   263→          accent="from-rose-500 to-red-600"
   264→        />
   265→        <SummaryCard
   266→          label="Ingresos"
   267→          value={incomeCount}
   268→          hint="Tipos de ingreso"
   269→          accent="from-amber-500 to-orange-600"
   270→        />
   271→        <SummaryCard
   272→          label="Subcategorías"
   273→          value={totalSubs}
   274→          hint="En total"
   275→          accent="from-violet-500 to-purple-600"
   276→        />
   277→      </div>
   278→    </div>
   279→  );
   280→}

   311→
   312→function CategoryCard({
   313→  category,
   314→  onUpdate,
   315→  onDelete,
   316→  onAddSubcategory,
   317→  onDeleteSubcategory,
   318→  onRenameSubcategory,
   319→}: {
   320→  category: Category;
   321→  onUpdate: (id: string, payload: { name: string; icon: string; color: string; type: string }) => void;
   322→  onDelete: (id: string, name: string) => void;
   323→  onAddSubcategory: (categoryId: string, name: string) => void;
   324→  onDeleteSubcategory: (id: string, name: string) => void;
   325→  onRenameSubcategory: (id: string, name: string) => void;
   326→}) {
   327→  const isIncome = category.type === "income";
   328→  const [editOpen, setEditOpen] = React.useState(false);
   329→  const [deleteOpen, setDeleteOpen] = React.useState(false);
   330→  const [newSub, setNewSub] = React.useState("");
   331→  const [addingSub, setAddingSub] = React.useState(false);
   332→  const [editingSubId, setEditingSubId] = React.useState<string | null>(null);
   333→  const [editingSubName, setEditingSubName] = React.useState("");
   334→
   335→  async function submitNewSub() {
   336→    const name = newSub.trim();
   337→    if (!name) return;
   338→    setAddingSub(true);
   339→    await onAddSubcategory(category.id, name);
   340→    setAddingSub(false);
   341→    setNewSub("");
   342→  }
   343→
   344→  async function submitRenameSub() {
   345→    const name = editingSubName.trim();
   346→    if (!name || !editingSubId) {
   347→      setEditingSubId(null);
   348→      return;
   349→    }
   350→    await onRenameSubcategory(editingSubId, name);
   351→    setEditingSubId(null);
   352→    setEditingSubName("");
   353→  }
   354→
   355→  return (
   356→    <AccordionItem
   357→      value={category.id}
   358→      className="rounded-xl border bg-card shadow-sm overflow-hidden border-b"
   359→    >
   360→      <div className="flex items-start gap-3 p-4">
   361→        <CategoryIcon
   362→          icon={category.icon}
   363→          color={category.color}
   364→          size="lg"
   365→          className="shrink-0"
   366→        />
   367→        <div className="flex-1 min-w-0">
   368→          <div className="flex items-center justify-between gap-2">
   369→            <p className="font-semibold truncate">{category.name}</p>
   370→            <div className="flex items-center gap-0.5">
   371→              <button
   372→                type="button"
   373→                onClick={() => setEditOpen(true)}
   374→                className="text-muted-foreground hover:text-foreground transition-colors p-1 rounded-md hover:bg-accent"
   375→                aria-label="Editar categoría"
   376→                title="Editar"
   377→              >
   378→                <Pencil className="h-3.5 w-3.5" />
   379→              </button>
   380→              <button
   381→                type="button"
   382→                onClick={() => setDeleteOpen(true)}
   383→                className="text-muted-foreground hover:text-red-500 transition-colors p-1 rounded-md hover:bg-accent"
   384→                aria-label="Eliminar categoría"
   385→                title="Eliminar"
   386→              >
   387→                <Trash2 className="h-3.5 w-3.5" />
   388→              </button>
   389→            </div>
   390→          </div>
   391→          <div className="flex items-center gap-2 mt-1.5 flex-wrap">
   392→            <Badge
   393→              variant="secondary"
   394→              className={cn(
   395→                "gap-1",
   396→                isIncome
   397→                  ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 hover:bg-emerald-500/10"
   398→                  : "bg-rose-500/10 text-rose-600 dark:text-rose-400 hover:bg-rose-500/10"
   399→              )}
   400→            >
   401→              {isIncome ? (
   402→                <ArrowDownRight className="h-3 w-3" />
   403→              ) : (
   404→                <ArrowUpRight className="h-3 w-3" />
   405→              )}
   406→              {isIncome ? "Ingreso" : "Gasto"}
   407→            </Badge>
   408→            <Badge variant="outline" className="gap-1">
   409→              <Hash className="h-3 w-3" />
   410→              {category.subcategories.length} sub
   411→            </Badge>
   412→          </div>
   413→        </div>
   414→      </div>
   415→      <AccordionTrigger className="px-4 pb-3 pt-0 hover:no-underline text-xs text-muted-foreground">
   416→        <span>
   417→          {category.subcategories.length > 0
   418→            ? "Ver subcategorías"
   419→            : "Sin subcategorías"}
   420→        </span>
   421→      </AccordionTrigger>
   422→      <AccordionContent className="px-4 pb-4">
   423→        <div className="space-y-2">
   424→          {category.subcategories.length === 0 && (
   425→            <p className="text-xs text-muted-foreground py-2">
   426→              Aún no hay subcategorías.
   427→            </p>
   428→          )}
   429→          {category.subcategories.map((s) => (
   430→            <div
   431→              key={s.id}
   432→              className="flex items-center gap-2 rounded-md border bg-muted/30 px-2.5 py-1.5 text-sm"
   433→            >
   434→              <ChevronRight className="h-3 w-3 text-muted-foreground" />
   435→              {editingSubId === s.id ? (
   436→                <Input
   437→                  value={editingSubName}
   438→                  onChange={(e) => setEditingSubName(e.target.value)}
   439→                  onKeyDown={(e) => {
   440→                    if (e.key === "Enter") submitRenameSub();
   441→                    if (e.key === "Escape") setEditingSubId(null);
   442→                  }}
   443→                  className="h-6 text-sm px-1.5 py-0"
   444→                  autoFocus
   445→                />
   446→              ) : (
   447→                <span className="flex-1 truncate">{s.name}</span>
   448→              )}
   449→              {editingSubId === s.id ? (
   450→                <>
   451→                  <button
   452→                    type="button"
   453→                    onClick={submitRenameSub}
   454→                    className="text-emerald-600 hover:text-emerald-700 p-1"
   455→                    aria-label="Guardar"
   456→                  >
   457→                    <Check className="h-3.5 w-3.5" />
   458→                  </button>
   459→                  <button
   460→                    type="button"
   461→                    onClick={() => setEditingSubId(null)}
   462→                    className="text-muted-foreground hover:text-foreground p-1"
   463→                    aria-label="Cancelar"
   464→                  >
   465→                    <X className="h-3.5 w-3.5" />
   466→                  </button>
   467→                </>
   468→              ) : (
   469→                <>
   470→                  <button
   471→                    type="button"
   472→                    onClick={() => {
   473→                      setEditingSubId(s.id);
   474→                      setEditingSubName(s.name);
   475→                    }}
   476→                    className="text-muted-foreground hover:text-foreground p-1"
   477→                    aria-label="Renombrar"
   478→                    title="Renombrar"
   479→                  >
   480→                    <Pencil className="h-3 w-3" />
   481→                  </button>
   482→                  <button
   483→                    type="button"
   484→                    onClick={() => onDeleteSubcategory(s.id, s.name)}
   485→                    className="text-muted-foreground hover:text-red-500 p-1"
   486→                    aria-label="Eliminar subcategoría"
   487→                    title="Eliminar"
   488→                  >
   489→                    <Trash2 className="h-3 w-3" />
   490→                  </button>
   491→                </>
   492→              )}
   493→            </div>
   494→          ))}
   495→          {/* Input para agregar subcategoría */}
   496→          <div className="flex gap-1.5 mt-1">
   497→            <Input
   498→              value={newSub}
   499→              onChange={(e) => setNewSub(e.target.value)}
   500→              onKeyDown={(e) => {
   501→                if (e.key === "Enter") submitNewSub();
   502→              }}
   503→              placeholder="Nueva subcategoría..."
   504→              className="h-8 text-xs"
   505→            />
   506→            <Button
   507→              type="button"
   508→              variant="outline"
   509→              size="sm"
   510→              className="h-8 px-2 gap-1 text-xs shrink-0"
   511→              onClick={submitNewSub}
   512→              disabled={!newSub.trim() || addingSub}
   513→            >
   514→              <Plus className="h-3.5 w-3.5" />
   515→              Agregar
   516→            </Button>
   517→          </div>
   518→        </div>
   519→      </AccordionContent>
   520→
   521→      {/* Dialog de edición */}
   522→      <EditCategoryDialog
   523→        open={editOpen}
   524→        onOpenChange={setEditOpen}
   525→        category={category}
   526→        onSubmit={(payload) => {
   527→          onUpdate(category.id, payload);
   528→          setEditOpen(false);
   529→        }}
   530→      />
   531→
   532→      {/* Confirmación de eliminación */}
   533→      <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   534→        <AlertDialogContent>
   535→          <AlertDialogHeader>
   536→            <AlertDialogTitle>¿Eliminar "{category.name}"?</AlertDialogTitle>
   537→            <AlertDialogDescription>
   538→              Esta acción no se puede deshacer. Si hay gastos asociados a esta
   539→              categoría, no se podrá eliminar hasta reasignarlos.
   540→            </AlertDialogDescription>
   541→          </AlertDialogHeader>
   542→          <AlertDialogFooter>
   543→            <AlertDialogCancel>Cancelar</AlertDialogCancel>
   544→            <AlertDialogAction
   545→              className="bg-red-600 hover:bg-red-700"
   546→              onClick={() => {
   547→                onDelete(category.id, category.name);
   548→                setDeleteOpen(false);
   549→              }}
   550→            >
   551→              <Trash2 className="h-4 w-4" />
   552→              Eliminar
   553→            </AlertDialogAction>
   554→          </AlertDialogFooter>
   555→        </AlertDialogContent>
   556→      </AlertDialog>
   557→    </AccordionItem>
   558→  );
   559→}
   560→
   561→function AddCategoryDialog({
   562→  open,
   563→  onOpenChange,
   564→  onSubmit,
   565→}: {
   566→  open: boolean;
   567→  onOpenChange: (v: boolean) => void;
   568→  onSubmit: (payload: {
   569→    name: string;
   570→    icon: string;
   571→    color: string;
   572→    type: string;
   573→  }) => void;
   574→}) {
   575→  const [name, setName] = React.useState("");
   576→  const [icon, setIcon] = React.useState("Wallet");
   577→  const [color, setColor] = React.useState("emerald");
   578→  const [type, setType] = React.useState("expense");
   579→  const [submitting, setSubmitting] = React.useState(false);
   580→
   581→  function reset() {
   582→    setName("");
   583→    setIcon("Wallet");
   584→    setColor("emerald");
   585→    setType("expense");
   586→  }
   587→
   588→  async function handleSubmit(e: React.FormEvent) {
   589→    e.preventDefault();
   590→    if (!name.trim()) {
   591→      toast.error("El nombre es obligatorio");
   592→      return;
   593→    }
   594→    setSubmitting(true);
   595→    await onSubmit({
   596→      name: name.trim(),
   597→      icon,
   598→      color,
   599→      type,
   600→    });
   601→    setSubmitting(false);
   602→    reset();
   603→  }
   604→
   605→  return (
   606→    <>
   607→      <Button className="gap-2" onClick={() => onOpenChange(true)}>
   608→        <Plus className="h-4 w-4" />
   609→        Agregar categoría

   643→function EditCategoryDialog({
   644→  open,
   645→  onOpenChange,
   646→  category,
   647→  onSubmit,
   648→}: {
   649→  open: boolean;
   650→  onOpenChange: (v: boolean) => void;
   651→  category: Category;
   652→  onSubmit: (payload: { name: string; icon: string; color: string; type: string }) => void;
   653→}) {
   654→  const [name, setName] = React.useState(category.name);
   655→  const [icon, setIcon] = React.useState(category.icon);
   656→  const [color, setColor] = React.useState(category.color);
   657→  const [type, setType] = React.useState(category.type);
   658→  const [submitting, setSubmitting] = React.useState(false);
   659→
   660→  React.useEffect(() => {
   661→    if (open) {
   662→      setName(category.name);
   663→      setIcon(category.icon);
   664→      setColor(category.color);
   665→      setType(category.type);
   666→    }
   667→  }, [open, category]);
   668→
   669→  async function handleSubmit() {
   670→    if (!name.trim()) {
   671→      toast.error("El nombre es obligatorio");
   672→      return;
   673→    }
   674→    setSubmitting(true);
   675→    await onSubmit({
   676→      name: name.trim(),
   677→      icon,
   678→      color,
   679→      type,
   680→    });
   681→    setSubmitting(false);
   682→  }
   683→
   684→  return (
   685→    <ModalContainer open={open} onOpenChange={onOpenChange} maxWidth="sm:max-w-lg">
   686→      <div className="px-6 pt-6 pb-3 shrink-0">
   687→        <h2 className="text-base font-semibold">Editar categoría</h2>
   688→        <p className="text-sm text-muted-foreground mt-1">
   689→          Modifica el nombre, ícono, color o tipo.
   690→        </p>
   691→      </div>
   692→      <div className="px-6 pb-6 space-y-4 overflow-y-auto scrollbar-thin">
   693→        <CategoryFormFields
   694→          name={name} setName={setName}
   695→          icon={icon} setIcon={setIcon}
   696→          color={color} setColor={setColor}
   697→          type={type} setType={setType}
   698→        />
   699→      </div>
   700→      <div className="flex gap-2 px-6 pb-6 pt-2 shrink-0 border-t mt-2">
   701→        <Button
   702→          type="button"
   703→          variant="outline"
   704→          onClick={() => onOpenChange(false)}
   705→        >
   706→          Cancelar
   707→        </Button>
   708→        <Button type="button" disabled={submitting} onClick={handleSubmit}>
   709→          {submitting ? "Guardando…" : "Guardar cambios"}
   710→        </Button>
   711→      </div>
   712→    </ModalContainer>
   713→  );
   714→}
